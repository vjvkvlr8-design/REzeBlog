// 페이지 로드 시 자동 실행
'use client'
import { usePathname } from 'next/navigation'
import { useEffect, useRef } from 'react'

export function VisitorTracker() {
  const pathname = usePathname()
  const startTime = useRef(Date.now())
  const pages = useRef<string[]>([])

  useEffect(() => {
    pages.current.push(pathname)
    
    // 페이지 이탈 시 데이터 전송
    const sendData = () => {
      navigator.sendBeacon('/api/visitor', JSON.stringify({
        referrer: document.referrer,
        pages: pages.current,
        duration: Math.floor((Date.now() - startTime.current) / 1000),
      }))
    }
    
    window.addEventListener('beforeunload', sendData)
    return () => window.removeEventListener('beforeunload', sendData)
  }, [pathname])

  return null // UI 없음
}

5-2. 서버 API (app/api/visitor/route.ts)

// POST /api/visitor
export async function POST(request: Request) {
  const data = await request.json()
  const ip = request.headers.get('x-forwarded-for') || 'unknown'
  
  // GeoIP로 국가 판별 (vercel은 x-vercel-ip-country 헤더 제공)
  const country = request.headers.get('x-vercel-ip-country') || '??'
  
  // 키워드 추출: referrer에서 q= 파라미터 파싱
  let keyword = null
  if (data.referrer) {
    const url = new URL(data.referrer)
    keyword = url.searchParams.get('q') || url.searchParams.get('query')
  }
  
  await prisma.visitor.create({
    data: { ip, country, referrer: data.referrer,
            keyword, pages: data.pages, duration: data.duration }
  })
}

7-1. 인증 체크 유틸 (lib/admin-auth.ts)

import { cookies } from 'next/headers'

export function isAuthenticated(): boolean {
  const token = cookies().get('admin_token')?.value
  if (!token) return false
  try {
    const decoded = Buffer.from(token, 'base64').toString()
    const [prefix, , pass] = decoded.split(':')
    return prefix === 'admin' && pass === process.env.ADMIN_PASSWORD
  } catch { return false }
}

// app/api/admin/posts/route.ts
import { isAuthenticated } from '@/lib/admin-auth'
import { prisma } from '@/lib/db'

// 게시글 생성
export async function POST(request: Request) {
  if (!isAuthenticated()) return unauthorized()
  const { title, content, channelId, author } = await request.json()
  const slug = title.toLowerCase()
    .replace(/[^가-힣a-z0-9\s-]/g, '')
    .replace(/\s+/g, '-')
  
  const post = await prisma.post.create({
    data: { title, slug, content, channelId, author }
  })
  return NextResponse.json(post)
}
